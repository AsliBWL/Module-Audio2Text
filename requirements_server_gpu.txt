# requirements_server_gpu.txt
# Файл зависимостей для запуска удалённого сервера whisper-live на ПК с GPU
# Установка: pip install -r requirements_server_gpu.txt

# =====================================================
# Основные библиотеки
# =====================================================

# NumPy - работа с многомерными массивами
numpy>=1.24.0,<2.0.0

# =====================================================
# Сервер whisper-live
# =====================================================

# whisper-live - серверная часть для удалённого распознавания речи
# Обеспечивает WebSocket-сервер, который принимает аудио и возвращает текст
# Установка: pip install whisper-live
whisper-live>=1.0.0

# faster-whisper - оптимизированная версия Whisper (используется сервером)
# Преимущества: в 4-5 раз быстрее оригинального Whisper
faster-whisper>=1.0.0

# =====================================================
# PyTorch с поддержкой CUDA (для GPU)
# =====================================================

# ВАЖНО: Установите PyTorch с поддержкой CUDA перед установкой остальных зависимостей:
# pip install torch torchaudio --index-url https://download.pytorch.org/whl/cu118

# =====================================================
# Дополнительные зависимости сервера
# =====================================================

# websockets - для WebSocket-соединений
websockets>=10.0

# =====================================================
# Примечания по установке
# =====================================================

# 1. Убедитесь, что установлен PyTorch с CUDA (см. команду выше)
# 2. Для работы whisper-live требуется:
#    - CUDA Toolkit >= 11.8
#    - NVIDIA драйвер >= 520.61
# 3. Проверка установки CUDA:
#    python -c "import torch; print(torch.cuda.is_available())"
# 4. Запуск сервера:
#    python -m whisper_live.server
