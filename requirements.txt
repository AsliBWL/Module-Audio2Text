# requirements.txt
# Файл зависимостей для проекта распознавания речи с голосовым помощником
# Установка: pip install -r requirements.txt

# =====================================================
# Основные библиотеки
# =====================================================

# NumPy - работа с многомерными массивами и математическими операциями
# Используется для: хранения и обработки аудиоданных, преобразований
numpy>=1.24.0,<2.0.0

# =====================================================
# Распознавание речи (STT - Speech-to-Text)
# =====================================================

# faster-whisper - оптимизированная версия Whisper от OpenAI
# Используется для: преобразования аудио в текст
# Преимущества: в 4-5 раз быстрее оригинального Whisper
faster-whisper>=1.0.0

# =====================================================
# Детекция голосовой активности (VAD)
# =====================================================

# silero-vad - нейросетевая модель для обнаружения границ речи
# Используется для: определения начала и конца речевых сегментов
# Преимущества: высокая точность, маленький размер (~2 МБ)
silero-vad>=6.0.0

# onnxruntime - оптимизированный рантайм для ONNX-моделей
# Используется для: ускоренной работы Silero VAD
# ONNX формат даёт прирост скорости по сравнению с PyTorch
onnxruntime>=1.14.0

# =====================================================
# Захват и обработка аудио
# =====================================================

# sounddevice - библиотека для захвата и воспроизведения звука
# Используется для: получения аудио с микрофона в реальном времени
sounddevice>=0.4.6

# torchaudio - библиотека для работы с аудио от PyTorch
# Используется для: загрузки аудиофайлов, ресемплинга, преобразований
torchaudio>=2.0.0

# torch - основная библиотека машинного обучения (зависимость torchaudio)
# Автоматически устанавливается вместе с torchaudio
torch>=2.0.0

# typing - встроенный модуль для аннотаций типов
