# =====================================================

# =====================================================

# faster-whisper - распознавание речи (используется для локального распознавания)

faster-whisper>=1.0.0

# Для CPU рекомендуется onnxruntime (без -gpu)

# =====================================================

# sounddevice - захват аудио с микрофона в реальном времени

# =====================================================

# =====================================================
# Примечания по установке
# =====================================================

# 1. Для работы RemoteWhisperRecognizer требуется SoX (системная программа):
#    - Windows: скачать с http://sox.sourceforge.net/ и добавить в PATH
#    - Linux: sudo apt-get install sox libsndfile1
#    - macOS: brew install sox

# 2. Для работы звукового ввода требуется PortAudio:
#    - Windows: скачать с https://www.portaudio.com/
#    - Linux: sudo apt-get install portaudio19-dev
#    - macOS: brew install portaudio

# 3. Для удалённого распознавания настройте Config.server_host в коде:
#    config.server_host = "IP-адрес_сервера_whisper-live"

# 4. В файле context.py раскомментируйте строку:
#    self.recognizer = RemoteWhisperRecognizer(self.config)
#    и закомментируйте:
#    self.recognizer = FasterWhisperRecognizer(self.config)
